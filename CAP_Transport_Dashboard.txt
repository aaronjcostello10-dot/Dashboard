# Ireland Climate Action Plan - Transport Dashboard
# Avoid-Shift-Improve Framework — all data pulled live from CSO PxStat

# install.packages(c("shiny", "shinydashboard", "plotly", "dplyr", "tidyr", "scales", "csodata"))
library(shiny)
library(shinydashboard)
library(plotly)
library(dplyr)
library(tidyr)
library(scales)
library(csodata)

# ============================================================================
# CSO DATA
# ============================================================================

get_actual_cso_data <- function() {

  # Energy consumption by sector (ktoe) — Source: SEAI SEI06
  # SEI06's Fuel.Type dimension is a hierarchy: individual fuels, running
  # subtotals (e.g. "Sum of all coal products"), and a grand total ("Sum of
  # all fuel products"). Summing across all Fuel.Type rows would multiply-count
  # each sector, so keep only the grand-total row per sector/year. Likewise
  # the whole-economy total is a Sector value ("Final energy consumption"),
  # not a Statistic value — SEI06's Statistic column only ever takes one value.
  sectors_of_interest <- c(
    "Sum of all transport", "Residential", "Sum of all industry sub-sectors",
    "Commercial services", "Public services", "Agricultural", "Fisheries"
  )

  sei06_long <- cso_get_data("SEI06") %>%
    filter(Fuel.Type == "Sum of all fuel products",
           Sector %in% c("Final energy consumption", sectors_of_interest)) %>%
    pivot_longer(
      cols = -any_of(c("Statistic", "Sector", "Fuel.Type")),
      names_to = "Year",
      values_to = "ktoe"
    ) %>%
    mutate(Year = as.integer(Year)) %>%
    filter(!is.na(ktoe) & Year >= 2018 & Year <= 2024)

  total_by_year <- sei06_long %>%
    filter(Sector == "Final energy consumption") %>%
    select(Year, Total = ktoe)

  energy_by_sector <- sei06_long %>%
    filter(Sector %in% sectors_of_interest) %>%
    select(Year, Sector, ktoe) %>%
    pivot_wider(names_from = Sector, values_from = ktoe) %>%
    left_join(total_by_year, by = "Year") %>%
    mutate(Transport_Share_Pct = round(100 * `Sum of all transport` / Total, 1)) %>%
    arrange(Year)
  
  # Remote Working Data - FETCH FROM CSO QES20
  # Source: CSO Table QES20 - Persons in employment - Extent working at home
  qes20_raw <- cso_get_data("QES20")

  # Get all employment baseline
  all_employment <- qes20_raw %>%
    filter(Extent.of.working.at.home == "All persons in employment")

  # Get remote workers (usually + sometimes)
  remote_workers <- qes20_raw %>%
    filter(Extent.of.working.at.home %in% c("Person usually works at home",
                                            "Person sometimes works at home"))

  all_employment_long <- all_employment %>%
    pivot_longer(
      cols = -c(Statistic, Extent.of.working.at.home),
      names_to = "Quarter",
      values_to = "Total_Employment"
    ) %>%
    mutate(Year = as.integer(substr(Quarter, 1, 4)))

  remote_workers_long <- remote_workers %>%
    pivot_longer(
      cols = -c(Statistic, Extent.of.working.at.home),
      names_to = "Quarter",
      values_to = "Number_Remote"
    ) %>%
    mutate(Year = as.integer(substr(Quarter, 1, 4)))

  # Calculate annual averages, then the remote-work share
  remote_annual <- remote_workers_long %>%
    group_by(Year, Quarter) %>%
    summarise(Total_Remote = sum(Number_Remote, na.rm = TRUE), .groups = 'drop') %>%
    group_by(Year) %>%
    summarise(Avg_Remote = mean(Total_Remote, na.rm = TRUE), .groups = 'drop')

  total_annual <- all_employment_long %>%
    group_by(Year) %>%
    summarise(Avg_Employed = mean(Total_Employment, na.rm = TRUE), .groups = 'drop')

  remote_work <- remote_annual %>%
    left_join(total_annual, by = "Year") %>%
    mutate(
      Remote_Work_Percent = round((Avg_Remote / Avg_Employed) * 100, 1)
    ) %>%
    filter(Year >= 2016 & Year <= 2024) %>%
    arrange(Year)

  # Private Car VKT — Source: CSO Table THA10 - Road Traffic Volumes by Type of Vehicle
  # THA10 comes in wide format (Statistic, Type.of.Vehicle, then one column per year)
  private_car_vkt <- cso_get_data("THA10") %>%
    filter(Type.of.Vehicle == "Private cars") %>%
    pivot_longer(
      cols = -c(Statistic, Type.of.Vehicle),
      names_to = "Year",
      values_to = "VALUE"
    ) %>%
    mutate(Year = as.integer(Year)) %>%
    filter(Year >= 2016 & Year <= 2023) %>%
    select(Year, Statistic, VALUE) %>%
    pivot_wider(names_from = Statistic, values_from = VALUE) %>%
    rename(
      Vehicle_Population = `Vehicle Population (Number)`,
      Total_Million_Km = `Kilometres Travelled (Million)`,
      Average_Km = `Average Kilometres Travelled (Kilometres)`
    ) %>%
    mutate(
      Total_Billion_Km = Total_Million_Km / 1000,
      Vehicle_Population_Million = Vehicle_Population / 1000000,
      Change_vs_2019_Pct = round(((Total_Million_Km / Total_Million_Km[Year == 2019]) - 1) * 100, 1),
      Change_vs_2018_Pct = round(((Total_Million_Km / Total_Million_Km[Year == 2018]) - 1) * 100, 1)
    ) %>%
    arrange(Year)

  # Road Freight Activity — Source: CSO Table TFQ01 - Road Freight Transport Activity
  tfq01_raw <- cso_get_data("TFQ01")

  # Helper: pivot one statistic to annual totals
  annual_stat <- function(df, stat_name, col_name) {
    df %>%
      filter(Statistic == stat_name,
             Business.of.Owner == "All business activities") %>%
      pivot_longer(
        cols      = -c(Statistic, Business.of.Owner),
        names_to  = "Quarter",
        values_to = "Value"
      ) %>%
      mutate(Year = as.integer(substr(Quarter, 1, 4))) %>%
      group_by(Year) %>%
      summarise(!!col_name := round(sum(Value, na.rm = TRUE), 0),
                .groups = "drop") %>%
      filter(Year >= 2016 & Year <= 2024)
  }

  tonne_km <- annual_stat(tfq01_raw, "Tonne-Kilometre",    "Tonne_Km_Million")
  km_trav  <- annual_stat(tfq01_raw, "Kilometres Travelled", "KM_Travelled_Million")

  road_freight <- full_join(tonne_km, km_trav, by = "Year") %>% arrange(Year)

  # Road Freight Tonne-Km by NST 2007 goods group — Source: CSO Table TFA19
  nst_groups <- c(
    "Products of agriculture, forestry and fishing",
    "Quarry products, metal ores and peat",
    "Foodstuffs",
    "Other non-metallic mineral products"
  )
  
  tfa19_raw <- cso_get_data("TFA19")

  # Identify the goods/NST dimension column (not Statistic — its name varies by release)
  goods_col <- names(tfa19_raw)[grepl("Goods|NST|Commodity|Type", names(tfa19_raw), ignore.case = TRUE) &
                                  names(tfa19_raw) != "Statistic"][1]

  freight_nst <- tfa19_raw %>%
    filter(grepl("Tonne.Kilometre|Tonne-Kilometre", Statistic, ignore.case = TRUE)) %>%
    filter(.data[[goods_col]] %in% nst_groups) %>%
    pivot_longer(
      cols      = -c(Statistic, all_of(goods_col)),
      names_to  = "Year",
      values_to = "Tonne_Km_Million"
    ) %>%
    mutate(
      Year      = as.integer(Year),
      NST_Group = .data[[goods_col]]
    ) %>%
    filter(!is.na(Tonne_Km_Million), Year >= 2014) %>%
    select(Year, NST_Group, Tonne_Km_Million) %>%
    arrange(NST_Group, Year)

  # Road Freight Tonne-Km by Main Use — Source: CSO Table TFA01
  main_use_groups <- c(
    "Other work",
    "Import/Export work",
    "Delivery of goods to retail outlets",
    "Delivery of goods to households",
    "Delivery of goods to wholesalers",
    "Delivery of materials and fuels to factories",
    "Delivery of goods to road works or building sites",
    "Carriage of livestock",
    "Carriage of other farm produce from farms",
    "Carriage of fertilisers, feeding stuffs etc. to farms"
  )
  
  tfa01_raw <- cso_get_data("TFA01")

  # Identify the main use dimension column (its name varies by release)
  use_col <- names(tfa01_raw)[grepl("Use|Type|Activity|Main", names(tfa01_raw), ignore.case = TRUE) &
                                names(tfa01_raw) != "Statistic"][1]

  freight_main_use <- tfa01_raw %>%
    filter(grepl("Tonne.Kilometre|Tonne-Kilometre", Statistic, ignore.case = TRUE)) %>%
    filter(.data[[use_col]] %in% main_use_groups) %>%
    pivot_longer(
      cols      = -c(Statistic, all_of(use_col)),
      names_to  = "Year",
      values_to = "Tonne_Km_Million"
    ) %>%
    mutate(
      Year      = as.integer(Year),
      Main_Use  = .data[[use_col]]
    ) %>%
    filter(!is.na(Tonne_Km_Million), Year >= 2014) %>%
    select(Year, Main_Use, Tonne_Km_Million) %>%
    arrange(Main_Use, Year)

  # ============================================================================
  # SHIFT TAB DATA FETCHES
  # ============================================================================

  # Bus passenger services — Source: CSO Table TOA16
  # Raw values are in THOUSANDS, not passenger counts — divide by 1e3 for millions.
  pt_bus_services <- cso_get_data("TOA16") %>%
    filter(Statistic == "Passengers") %>%
    pivot_longer(
      cols = -c(Statistic, Scheduled.services),
      names_to = "Year", values_to = "Passengers_Millions"
    ) %>%
    mutate(
      Year = as.integer(Year),
      Passengers_Millions = round(Passengers_Millions / 1e3, 2)
    ) %>%
    rename(Service_Type = Scheduled.services) %>%
    filter(!is.na(Passengers_Millions)) %>%
    arrange(Service_Type, Year)

  # Rail passenger journeys — Source: CSO Table THA24 (has Year AND Weeks-of-year; aggregate across weeks)
  # Rail was only added to THA24 from 2020 onwards, so 2019 rail sums to 0 —
  # drop zero-total rows so downstream charts don't show false zeros.
  pt_rail_annual <- cso_get_data("THA24") %>%
    filter(Statistic == "Passenger Journeys") %>%
    pivot_longer(
      cols = -c(Statistic, Mode.of.Transport, Weeks.of.the.year),
      names_to = "Year", values_to = "Journeys"
    ) %>%
    mutate(Year = as.integer(Year)) %>%
    filter(!is.na(Journeys)) %>%
    group_by(Year, Mode = Mode.of.Transport) %>%
    summarise(Journeys_Millions = round(sum(Journeys, na.rm = TRUE) / 1e6, 2),
              .groups = "drop") %>%
    filter(Journeys_Millions > 0) %>%   # drop rows where mode has no coverage
    arrange(Mode, Year)

  # Luas passenger numbers — Source: CSO Table TOA11
  # csodata returns months pivoted into columns like '2018_January', ... plus
  # a '..._All months' summary row per year, which we must drop to avoid double-counting.
  pt_luas_annual <- cso_get_data("TOA11") %>%
    filter(Statistic %in% c("Red line", "Green line", "All Luas lines")) %>%
    pivot_longer(
      cols = -Statistic,
      names_to = "Period", values_to = "Journeys"
    ) %>%
    filter(!grepl("All months", Period)) %>%
    mutate(Year = as.integer(substr(Period, 1, 4))) %>%
    filter(!is.na(Journeys), !is.na(Year)) %>%
    group_by(Year, Statistic) %>%
    summarise(Journeys_Millions = round(sum(Journeys, na.rm = TRUE) / 1e6, 2),
              .groups = "drop") %>%
    rename(Line = Statistic) %>%
    arrange(Line, Year)

  # Bike share journeys — Source: CSO Tables TOA08 (Dublin) and TOA22 (regional cities)
  toa08_raw <- cso_get_data("TOA08")
  dublin <- toa08_raw %>%
    filter(Statistic == "Journeys") %>%
    pivot_longer(
      cols = -Statistic,
      names_to = "Period", values_to = "Journeys"
    ) %>%
    filter(!grepl("All months", Period)) %>%
    mutate(Year = as.integer(substr(Period, 1, 4))) %>%
    filter(!is.na(Journeys), !is.na(Year)) %>%
    group_by(Year) %>%
    summarise(Journeys = sum(Journeys, na.rm = TRUE), .groups = "drop") %>%
    mutate(City = "Dublin")

  regional <- cso_get_data("TOA22") %>%
    filter(Statistic == "Journeys") %>%
    filter(Bicycle.Sharing.Scheme %in% c("Cork", "Galway", "Limerick", "Waterford")) %>%
    pivot_longer(
      cols = -c(Statistic, Bicycle.Sharing.Scheme),
      names_to = "Period", values_to = "Journeys"
    ) %>%
    mutate(Year = as.integer(substr(Period, 1, 4))) %>%
    filter(!is.na(Journeys)) %>%
    group_by(Year, City = Bicycle.Sharing.Scheme) %>%
    summarise(Journeys = sum(Journeys, na.rm = TRUE), .groups = "drop")

  bike_share <- bind_rows(dublin, regional) %>%
    mutate(Journeys_Thousands = round(Journeys / 1e3, 0)) %>%
    select(Year, City, Journeys_Thousands) %>%
    arrange(City, Year)

  # Census 2022 commute journey time — Source: CSO Table SAP2022T11T3ED
  sap_raw <- cso_get_data("SAP2022T11T3ED")
  # Geography and journey-time column names vary by release — find them dynamically
  geo_col <- names(sap_raw)[grepl("Electoral|Counties|State",
                                  names(sap_raw), ignore.case = TRUE)][1]
  time_col <- names(sap_raw)[grepl("Journey.Time", names(sap_raw), ignore.case = TRUE)][1]

  commute_journey_time <- sap_raw %>%
    filter(.data[[geo_col]] == "Ireland" | .data[[geo_col]] == "State") %>%
    filter(!.data[[time_col]] %in% c("Total", "Not stated")) %>%
    mutate(
      Journey_Time = .data[[time_col]],
      Count = `2022`
    ) %>%
    select(Journey_Time, Count) %>%
    filter(!is.na(Count)) %>%
    mutate(Pct = round(100 * Count / sum(Count, na.rm = TRUE), 1))

  # Private car indicators — Source: CSO Tables THA17 (vehicles & km) + PEA08 (population)
  tha17_raw <- cso_get_data("THA17")
  pea08_raw <- cso_get_data("PEA08")

  car_indicators <- {
    # THA17: national private cars, all fuel types, all registration years
    cars <- tha17_raw %>%
      filter(
        Fuel.Type             == "All fuel types",
        County.of.Ownership   == "Ireland",
        Type.of.Vehicle       == "Private cars",
        Year.of.Registration  == "All years"
      ) %>%
      select(Statistic, `2015`:`2023`) %>%
      pivot_longer(cols = -Statistic, names_to = "Year", values_to = "Value") %>%
      mutate(Year = as.integer(Year)) %>%
      pivot_wider(names_from = Statistic, values_from = Value) %>%
      rename(
        Vehicle_Population = `Vehicle Population`,
        KM_Travelled_M     = `Kilometres Travelled`,
        Average_KM         = `Average Kilometres Travelled`
      )
    
    # PEA08: national total population (thousands), supplemented with
    # 2015/2016 from SEAI/CSO source (Excel sheet: Private Car Traffic Activity)
    pop_early <- data.frame(
      Year            = c(2015L, 2016L),
      Population_000s = c(4687.8, 4739.6)
    )
    pop_pea08 <- pea08_raw %>%
      filter(Age.Group == "All ages", Sex == "Both sexes", County == "Ireland") %>%
      select(`2017`:`2023`) %>%
      pivot_longer(everything(), names_to = "Year", values_to = "Population_000s") %>%
      mutate(Year = as.integer(Year))
    pop <- bind_rows(pop_early, pop_pea08)
    
    # Join and derive indicators (left join keeps all THA17 years from 2015)
    combined <- cars %>%
      left_join(pop, by = "Year") %>%
      mutate(
        Population        = Population_000s * 1000,
        Cars_per_1000     = round(Vehicle_Population / Population * 1000, 1),
        KM_per_capita     = round(KM_Travelled_M * 1e6 / Population),
        # Index all variables to 2015=1
        base_veh = Vehicle_Population[Year == 2015],
        base_km  = KM_Travelled_M[Year == 2015],
        base_avg = Average_KM[Year == 2015],
        base_pop = Population[Year == 2015],
        Idx_Vehicle_Pop  = round(Vehicle_Population / base_veh, 4),
        Idx_KM_Travelled = round(KM_Travelled_M    / base_km,  4),
        Idx_Average_KM   = round(Average_KM        / base_avg, 4),
        Idx_Population   = round(Population        / base_pop, 4)
      ) %>%
      select(-starts_with("base_")) %>%
      arrange(Year)

    combined
  }

  # Freight vehicle indicators — Source: CSO Table THA17 (goods vehicles)
  # Vehicle Population, KM Travelled, Average KM, each indexed to 2015 = 1
  freight_vehicle_indicators <- {
    # THA17: national goods vehicles, all fuel types, all registration years
    goods <- tha17_raw %>%
      filter(
        Fuel.Type            == "All fuel types",
        County.of.Ownership  == "Ireland",
        Type.of.Vehicle      == "Goods vehicles",
        Year.of.Registration == "All years"
      ) %>%
      select(Statistic, `2015`:`2023`) %>%
      pivot_longer(cols = -Statistic, names_to = "Year", values_to = "Value") %>%
      mutate(Year = as.integer(Year)) %>%
      pivot_wider(names_from = Statistic, values_from = Value)
    
    # Rename columns to safe names
    names(goods) <- make.names(names(goods))
    goods <- goods %>% rename_with(~ gsub("\\.", "_", .x))
    
    # Identify the Vehicle Population, KM Travelled, Average KM columns
    veh_col <- grep("Vehicle_Population|Number_of_Vehicles|Vehicles_Licensed",
                    names(goods), value = TRUE, ignore.case = TRUE)[1]
    km_col  <- grep("Kilometre|Traveled|Travelled|Distance",
                    names(goods), value = TRUE, ignore.case = TRUE)[1]
    avg_col <- grep("Average",
                    names(goods), value = TRUE, ignore.case = TRUE)[1]
    
    goods %>%
      rename(Vehicle_Population = !!veh_col,
             KM_Travelled_M     = !!km_col,
             Average_KM         = !!avg_col) %>%
      mutate(
        base_veh = Vehicle_Population[Year == 2015],
        base_km  = KM_Travelled_M[Year == 2015],
        base_avg = Average_KM[Year == 2015],
        Idx_Vehicle_Pop  = round(Vehicle_Population / base_veh, 4),
        Idx_KM_Travelled = round(KM_Travelled_M     / base_km,  4),
        Idx_Average_KM   = round(Average_KM         / base_avg, 4)
      ) %>%
      select(-starts_with("base_")) %>%
      arrange(Year)
  }

  # Aviation activity index — Source: CSO Tables TAA04 (2016-2019) + TAA06 (2020-2024)
  # Passengers, Freight, Commercial Flights, indexed to 2019 = 100 (last pre-COVID year —
  # indexing to 2020 itself would overstate the post-pandemic recovery as "growth")
  taa04_raw <- cso_get_data("TAA04")
  taa06_raw <- cso_get_data("TAA06")

  aviation_activity <- {
    taa04_clean <- taa04_raw %>%
      filter(Airports.in.Ireland == "All main airports",
             Aircraft.Make       == "All makes of aircraft",
             Direction           == "All directions",
             Flight.Type         == "All flights") %>%
      select(Statistic, `2016`:`2019`) %>%
      pivot_longer(-Statistic, names_to = "Year", values_to = "Value") %>%
      mutate(Year = as.integer(Year))
    
    taa06_clean <- taa06_raw %>%
      filter(Airports.in.Ireland == "All main airports",
             Country             == "All Countries",
             Direction           == "All directions",
             Flight.Type         == "All flights") %>%
      select(Statistic, `2020`:`2024`) %>%
      pivot_longer(-Statistic, names_to = "Year", values_to = "Value") %>%
      mutate(Year = as.integer(Year))
    
    combined <- bind_rows(taa04_clean, taa06_clean) %>%
      arrange(Statistic, Year)
    
    # Index to 2019=100 (last pre-COVID year)
    base_2019 <- combined %>%
      filter(Year == 2019) %>%
      select(Statistic, Base = Value)
    
    combined %>%
      left_join(base_2019, by = "Statistic") %>%
      mutate(Index = round(Value / Base * 100, 1)) %>%
      select(Year, Statistic, Value, Index)
  }

  # Aviation passengers by airport — Source: CSO Table TAA05
  aviation_passengers <- cso_get_data("TAA05") %>%
    pivot_longer(cols = -c(Statistic, Airports.in.Ireland),
                 names_to = "Year", values_to = "Passengers") %>%
    mutate(
      Year  = as.integer(Year),
      Group = case_when(
        Airports.in.Ireland == "Dublin" ~ "Dublin",
        Airports.in.Ireland == "Cork"   ~ "Cork",
        TRUE                            ~ "Other"
      )
    ) %>%
    group_by(Year, Group) %>%
    summarise(Passengers = sum(Passengers, na.rm = TRUE), .groups = "drop") %>%
    arrange(Year, Group)

  # National GHG % by sector — Source: CSO Table BIIESGE03
  sectors_keep <- c("Agriculture", "Industry", "Transport", "Energy",
                    "Waste", "Commercial Services", "Public Services", "Residential")

  national_ghg_pct <- cso_get_data("BIIESGE03") %>%
    filter(Common.Reporting.Format.Sector %in% sectors_keep) %>%
    pivot_longer(
      cols = -c(Statistic, Common.Reporting.Format.Sector),
      names_to = "Year",
      values_to = "Pct"
    ) %>%
    mutate(Year = as.integer(Year), Sector = Common.Reporting.Format.Sector) %>%
    filter(Year >= 2018) %>%
    select(Year, Sector, Pct) %>%
    arrange(Year)

  # Commuting mode share — Source: CSO Table SAP2022T11T1ED (Census 2022, means of travel to work)
  commuting_mode <- cso_get_data("SAP2022T11T1ED") %>%
    filter(
      CSO.Electoral.Divisions.2022 == "Ireland",
      Statistic == "Usually resident by means of travel to work",
      Means.of.Travel != "Total"
    ) %>%
    select(Mode = Means.of.Travel, Count = `2022`) %>%
    filter(!is.na(Count)) %>%
    mutate(Pct = round(100 * Count / sum(Count, na.rm = TRUE), 1)) %>%
    arrange(desc(Count))

  # ============================================================================
  # IMPROVE TAB DATA FETCHES
  # ============================================================================

  # New/secondhand private cars by fuel type — Source: CSO Table TEM27
  new_cars_fuel <- cso_get_data("TEM27") %>%
    filter(Licensing.Authority == "All licensing authorities") %>%
    filter(Type.of.Fuel != "All fuel types") %>%
    pivot_longer(
      cols = -c(Statistic, Licensing.Authority, Type.of.Fuel),
      names_to = "Month", values_to = "Count"
    ) %>%
    mutate(
      Year = as.integer(substr(Month, 1, 4)),
      Fuel = Type.of.Fuel,
      Category = Statistic
    ) %>%
    filter(!is.na(Count)) %>%
    group_by(Year, Category, Fuel) %>%
    summarise(Count = sum(Count, na.rm = TRUE), .groups = "drop") %>%
    arrange(Category, Fuel, Year)

  # Vehicles licensed for the first time, annual long trend — Source: CSO Table TEA01
  # NB: TEA01 uses uppercase STATISTIC column name, unlike most other CSO tables
  vehicles_licensed_long <- cso_get_data("TEA01") %>%
    filter(Taxation.Class %in% c("New Private Cars",
                                 "New Goods Vehicles",
                                 "New Motor Cycles",
                                 "Secondhand Private Cars")) %>%
    pivot_longer(
      cols = -c(STATISTIC, Taxation.Class),
      names_to = "Year", values_to = "Count"
    ) %>%
    mutate(Year = as.integer(Year), Type = Taxation.Class) %>%
    filter(!is.na(Count), Year >= 1997) %>%
    select(Year, Type, Count) %>%
    arrange(Type, Year)

  # Private car fleet by fuel type — Source: CSO Table THA17
  car_fleet_fuel <- tha17_raw %>%
    filter(Statistic == "Vehicle Population") %>%
    filter(Type.of.Vehicle == "Private cars") %>%
    filter(County.of.Ownership == "Ireland") %>%
    filter(Year.of.Registration == "All years") %>%
    filter(Fuel.Type != "All fuel types") %>%
    pivot_longer(
      cols = -c(Statistic, Fuel.Type, Type.of.Vehicle,
                County.of.Ownership, Year.of.Registration),
      names_to = "Year", values_to = "Count"
    ) %>%
    mutate(Year = as.integer(Year), Fuel = Fuel.Type) %>%
    filter(!is.na(Count)) %>%
    select(Year, Fuel, Count) %>%
    arrange(Fuel, Year)


  list(
    pt_bus_services  = pt_bus_services,
    pt_rail_annual   = pt_rail_annual,
    pt_luas_annual   = pt_luas_annual,
    bike_share       = bike_share,
    commute_journey_time = commute_journey_time,
    new_cars_fuel    = new_cars_fuel,
    vehicles_licensed_long = vehicles_licensed_long,

# ============================================================================
# SERVER LOGIC
# ============================================================================

server <- function(input, output, session) {
  
  # Load actual CSO data
  data <- get_actual_cso_data()
  
  # ========================================================================
  # SETTING THE SCENE TAB OUTPUTS
  # ========================================================================
  
  output$scene_energy_trend <- renderPlotly({
    sector_cols <- c(
      "Sum of all transport", "Residential", "Sum of all industry sub-sectors",
      "Commercial services", "Public services", "Agricultural", "Fisheries"
    )
    
    # Pull 2024 row and compute % of national total
    row_2024 <- data$energy_by_sector %>% filter(Year == 2024)
    total_2024 <- row_2024$Total
    
    chart_data <- data.frame(
      Sector = sector_cols,
      ktoe   = as.numeric(row_2024[, sector_cols])
    ) %>%
      mutate(
        Pct    = round(100 * ktoe / total_2024, 1),
        Colour = ifelse(Sector == "Sum of all transport", "#d32f2f", "#607D8B"),
        Label  = paste0(Pct, "%")
      ) %>%
      arrange(desc(Pct))   # highest bar first
    
    plot_ly(chart_data,
            x = ~Pct,
            y = ~reorder(Sector, Pct),
            type = 'bar',
            orientation = 'h',
            marker = list(color = ~Colour),
            text = ~Label,
            textposition = 'outside',
            hovertemplate = '<b>%{y}</b><br>%{x:.1f}% of national total<extra></extra>') %>%
      layout(
        xaxis = list(title = "% of national final energy consumption",
                     range = list(0, 50),
                     gridcolor = "#e8ecf4", gridwidth = 1, showline = FALSE, zeroline = FALSE),
        yaxis = list(title = "", showgrid = FALSE, showline = FALSE),
        showlegend = FALSE,
        margin = list(l = 220, r = 60)
      )
  })
  
  output$scene_energy_treemap <- renderPlotly({
    sector_cols <- c(
      "Sum of all transport", "Residential", "Sum of all industry sub-sectors",
      "Commercial services", "Public services", "Agricultural", "Fisheries"
    )
    
    row_2024   <- data$energy_by_sector %>% filter(Year == 2024)
    total_2024 <- row_2024$Total
    
    chart_data <- data.frame(
      Sector = sector_cols,
      ktoe   = as.numeric(row_2024[, sector_cols])
    ) %>%
      mutate(
        Pct    = round(100 * ktoe / total_2024, 1),
        Colour = ifelse(Sector == "Sum of all transport", "#d32f2f", "#607D8B")
      )
    
    plot_ly(
      type    = "treemap",
      labels  = chart_data$Sector,
      values  = chart_data$ktoe,
      parents = rep("Ireland", nrow(chart_data)),
      marker  = list(colors = chart_data$Colour),
      texttemplate = "<b>%{label}</b><br>%{percentRoot:.1%}",
      hovertemplate = "<b>%{label}</b><br>%{value:,.0f} ktoe<br>%{percentRoot:.1%} of national total<extra></extra>"
    ) %>%
      layout(margin = list(t = 10, b = 10))
  })
  
  output$scene_ghg_sectors <- renderPlotly({
    # Wilke: perceptually distinct palette — no two similar greys
    sector_colours <- c(
      "Agriculture"         = "#4dac26",   # green
      "Transport"           = "#d32f2f",   # red   (consistent across dashboard)
      "Energy"              = "#f1a340",   # amber
      "Industry"            = "#998ec3",   # purple
      "Residential"         = "#2196F3",   # blue
      "Commercial Services" = "#74add1",   # light blue  (distinct from Residential)
      "Public Services"     = "#a6611a",   # brown
      "Waste"               = "#80cdc1"    # teal
    )
    
    plot_ly(data$national_ghg_pct,
            x = ~Year, y = ~Pct, color = ~Sector,
            type = 'bar', colors = sector_colours,
            hovertemplate = '<b>%{fullData.name}</b>: %{y:.1f}%<extra></extra>') %>%
      layout(
        barmode   = 'stack',
        xaxis     = list(title = "Year", showgrid = FALSE, showline = FALSE, ticks = "outside"),
        yaxis     = list(title = "% of National GHG Emissions", range = list(0, 101),
                         gridcolor = "#e8ecf4", gridwidth = 1, showline = FALSE, zeroline = FALSE),
        hovermode = 'x unified',
        legend    = list(x = 1.01, y = 0.99)
      )
  })
  
  output$scene_commuting <- renderPlotly({
    # Wilke: colour should encode meaning, not just distinguish categories already
    # labelled on the axis. Use semantic groupings:
    #   car modes -> red  |  public transport -> blue  |  active travel -> green
    #   home/remote -> amber  |  other/not stated -> grey
    mode_colours <- c(
      "Car Driver"                  = "#c0392b",  # car — red
      "Car passenger"               = "#e57373",  # car (lighter)
      "Van"                         = "#a93226",  # car-adjacent — dark red
      "Motorcycle or scooter"       = "#e8a090",  # car-adjacent — pale red
      "Bus, minibus or coach"       = "#2196F3",  # public transport — blue
      "Train, DART or LUAS"         = "#1565C0",  # public transport — dark blue
      "On Foot"                     = "#388e3c",  # active — green
      "Bicycle"                     = "#4CAF50",  # active — lighter green
      "Work mainly at or from home" = "#f1a340",  # remote — amber
      "Other (incl. lorry)"         = "#9E9E9E",  # other — grey
      "Not stated"                  = "#BDBDBD"   # unknown — light grey
    )
    
    chart <- data$commuting_mode %>%
      mutate(
        Colour = mode_colours[as.character(Mode)],
        Label  = paste0(Pct, "%")
      )
    
    plot_ly(chart,
            x = ~Pct,
            y = ~reorder(Mode, Pct),
            type = 'bar',
            orientation = 'h',
            marker = list(color = ~Colour),
            text = ~Label,
            textposition = 'outside',
            hovertemplate = '<b>%{y}</b>: %{x:.1f}% of workers<extra></extra>') %>%
      layout(
        xaxis = list(title = "% of Workers", range = list(0, 65),
                     showgrid = FALSE, showline = FALSE, zeroline = FALSE),
        yaxis = list(title = "", showgrid = FALSE, showline = FALSE),
        showlegend = FALSE,
        margin = list(l = 220, r = 60)
      )
  })
  
  
  
  
  # ========================================================================
  # AVOID TAB OUTPUTS
  # ========================================================================
  
  output$avoid_vkt_total_plot <- renderPlotly({
    # CAP 2025 Target: 20% reduction in VKT by 2030 relative to the 2018 baseline
    # 2018 baseline (actual): 35.959 billion km
    # 2030 target:            80% of 2018 baseline = 28.77 billion km
    baseline_2018 <- 35.959
    target_2030   <- baseline_2018 * 0.80
    
    # Linear trajectory: 2018 baseline → 2030 target
    trajectory_years <- 2018:2030
    trajectory_vals  <- baseline_2018 + (target_2030 - baseline_2018) *
      (trajectory_years - 2018) / (2030 - 2018)
    
    plot_ly() %>%
      # CAP25 target trajectory (drawn first so actuals sit on top)
      add_trace(
        x = trajectory_years, y = trajectory_vals,
        type = 'scatter', mode = 'lines',
        name = 'CAP 2025 target pathway',
        line = list(color = '#00959f', dash = 'dash', width = 2),
        hovertemplate = '<b>%{x}</b> target: %{y:.2f} billion km<extra></extra>'
      ) %>%
      # 2018 baseline reference
      add_trace(
        x = c(2018, 2030), y = c(baseline_2018, baseline_2018),
        type = 'scatter', mode = 'lines',
        name = '2018 baseline',
        line = list(color = '#888', dash = 'dot', width = 1),
        hoverinfo = 'skip'
      ) %>%
      # Actual VKT (solid red line, primary series)
      add_trace(
        data = data$private_car_vkt,
        x = ~Year, y = ~Total_Billion_Km,
        type = 'scatter', mode = 'lines+markers',
        name = 'Actual VKT',
        line   = list(color = '#c0392b', width = 3),
        marker = list(size = 8, color = '#c0392b'),
        hovertemplate = '<b>%{x}</b> actual: %{y:.2f} billion km<extra></extra>'
      ) %>%
      # 2030 endpoint marker
      add_trace(
        x = 2030, y = target_2030,
        type = 'scatter', mode = 'markers+text',
        name = '2030 target',
        marker = list(size = 12, color = '#00959f', symbol = 'diamond'),
        text = paste0("2030 target\n", round(target_2030, 1), " Bn km"),
        textposition = 'middle right',
        textfont = list(color = '#00959f', size = 11),
        hovertemplate = '<b>2030 target</b>: %{y:.2f} billion km<extra></extra>',
        showlegend = FALSE
      ) %>%
      layout(
        xaxis = list(title = "Year", range = c(2017.5, 2031.5),
                     showgrid = FALSE, showline = FALSE, ticks = "outside", dtick = 2),
        yaxis = list(title = "Billion km",
                     range = c(20, 40),
                     gridcolor = "#e8ecf4", gridwidth = 1,
                     showline = FALSE, zeroline = FALSE),
        hovermode = 'x unified',
        legend    = list(orientation = "h", x = 0, y = -0.22),
        annotations = list(
          list(
            x = 2018, y = baseline_2018,
            text = "2018 baseline<br>35.96 Bn km",
            showarrow = TRUE, arrowhead = 2, arrowsize = 0.8,
            ax = -30, ay = -30,
            font = list(size = 10, color = '#555'),
            bgcolor = 'rgba(255,255,255,0.85)', borderpad = 3
          )
        )
      )
  })
  
  output$avoid_vkt_average_plot <- renderPlotly({
    # Wilke: area fill to zero is misleading when zero is not a meaningful baseline
    # (a car travelling 0 km is not a relevant reference point). Use a plain line.
    baseline_avg <- 16352
    yr <- range(data$private_car_vkt$Year)
    
    plot_ly() %>%
      add_trace(
        data = data$private_car_vkt,
        x = ~Year, y = ~Average_Km,
        type = 'scatter', mode = 'lines+markers',
        name = 'Average km per car',
        line   = list(color = '#e6861a', width = 2.5),
        marker = list(size = 7, color = '#e6861a'),
        hovertemplate = '<b>%{x}</b>: %{y:,.0f} km<extra></extra>'
      ) %>%
      add_trace(
        x = c(yr[1], yr[2]),
        y = c(baseline_avg, baseline_avg),
        type = 'scatter', mode = 'lines',
        name = '2019 pre-pandemic level',
        line = list(color = '#888', dash = 'dot', width = 1.5),
        hoverinfo = 'skip'
      ) %>%
      layout(

      ) %>%

        xaxis = list(title = "Year", dtick = 1, showgrid = FALSE, showline = FALSE, ticks = "outside"),
        yaxis = list(title = "Index (2019 = 100)",
                     gridcolor = "#e8ecf4", gridwidth = 1,
                     showline = FALSE, zeroline = FALSE),
        legend = list(orientation = "h", x = 0, y = -0.22),
        hovermode = "x unified"
      )
  })
  
  # 5. Bike share journeys (Dublin + regional cities) — stacked bar
  output$shift_bike_share_plot <- renderPlotly({
    colours <- c(
      "Dublin"    = "#405381",
      "Cork"      = "#00959f",
      "Galway"    = "#5BC1A5",
      "Limerick"  = "#FCBE72",
      "Waterford" = "#c0392b"
    )
    plot_ly(data$bike_share,
            x = ~Year, y = ~Journeys_Thousands,
            color = ~City, colors = colours,
            type = "bar",
            hovertemplate = "<b>%{fullData.name}</b><br>%{x}: %{y:,.0f}k journeys<extra></extra>") %>%
      layout(
        barmode = "stack",
        xaxis = list(title = "Year", dtick = 1, showgrid = FALSE, showline = FALSE, ticks = "outside"),
        yaxis = list(title = "Thousand journeys",
                     gridcolor = "#e8ecf4", gridwidth = 1,
                     showline = FALSE, zeroline = FALSE),
        legend = list(orientation = "h", x = 0, y = -0.22),
        hovermode = "x unified"
      )
  })
  
  # 6a. Census 2022 commute mode share — horizontal bar
  output$shift_commute_mode_plot <- renderPlotly({
    # Re-use commuting_mode dataset (already fetched for Scene tab)
    mode_colours <- c(
      "Car Driver"                  = "#c0392b",
      "Car passenger"               = "#e57373",
      "Van"                         = "#a93226",
      "Motorcycle or scooter"       = "#e8a090",
      "Bus, minibus or coach"       = "#2196F3",
      "Train, DART or LUAS"         = "#1565C0",
      "On Foot"                     = "#388e3c",
      "Bicycle"                     = "#4CAF50",
      "Work mainly at or from home" = "#f1a340",
      "Other (incl. lorry)"         = "#9E9E9E",
      "Not stated"                  = "#BDBDBD"
    )
    
    chart <- data$commuting_mode %>%
      mutate(
        Colour = mode_colours[as.character(Mode)],
        Label  = paste0(Pct, "%")
      )
    
    plot_ly(chart,
            x = ~Pct, y = ~reorder(Mode, Pct),
            type = "bar", orientation = "h",
            marker = list(color = ~Colour),
            text = ~Label, textposition = "outside",
            hovertemplate = "<b>%{y}</b>: %{x:.1f}%<extra></extra>") %>%
      layout(
        xaxis = list(title = "% of workers", range = c(0, 65),
                     showgrid = FALSE, showline = FALSE, zeroline = FALSE),
        yaxis = list(title = "", showgrid = FALSE, showline = FALSE),
        showlegend = FALSE,
        margin = list(l = 180, r = 60)
      )
  })
  
  # 6b. Census 2022 commute journey time distribution — horizontal bar
  output$shift_commute_time_plot <- renderPlotly({
    # Order journey time categories sensibly
    time_order <- c(
      "Under 15 mins",
      "1/4 hour - under 1/2 hour",
      "1/2 hour - under 3/4 hour",
      "3/4 hour - under 1 hour",
      "1 hour - under 1 1/2 hours",
      "1 1/2 hours and over"
    )
    
    chart <- data$commute_journey_time %>%
      filter(Journey_Time %in% time_order) %>%
      mutate(
        Journey_Time = factor(Journey_Time, levels = rev(time_order)),
        Label = paste0(Pct, "%"),
        # Colour short commutes green (active travel opportunity), long ones grey
        Colour = case_when(
          Journey_Time %in% c("Under 15 mins", "1/4 hour - under 1/2 hour") ~ "#4dac26",
          Journey_Time == "1/2 hour - under 3/4 hour" ~ "#f1a340",
          TRUE ~ "#8693b3"
        )
      )
    
    plot_ly(chart,
            x = ~Pct, y = ~Journey_Time,
            type = "bar", orientation = "h",
            marker = list(color = ~Colour),
            text = ~Label, textposition = "outside",
            hovertemplate = "<b>%{y}</b>: %{x:.1f}%<extra></extra>") %>%
      layout(
        xaxis = list(title = "% of commuters",
                     showgrid = FALSE, showline = FALSE, zeroline = FALSE),
        yaxis = list(title = "", showgrid = FALSE, showline = FALSE),
        showlegend = FALSE,
        margin = list(l = 180, r = 60)
      )
  })
  
  # ========================================================================
  # IMPROVE TAB OUTPUTS (6 metrics)
  # ========================================================================
  
  # Shared fuel-type colour palette so all IMPROVE charts stay consistent
  fuel_colours <- c(
    "Petrol"                                    = "#8693b3",
    "Diesel"                                    = "#405381",
    "Petrol and electric hybrid"                = "#f1a340",
    "Diesel and electric hybrid"                = "#c0392b",
    "Petrol or Diesel plug-in hybrid electric"  = "#00959f",
    "Electric"                                  = "#4dac26",
    "Other fuel types"                          = "#BDBDBD"
  )
  
  # 1. New private cars by fuel type (TEM27) — stacked bar
  # Exclude 2026 (partial year — TEM27 only covers Jan-Mar 2026 so the bar
  # would be ~25% the height of full years and mislead the eye)
  output$improve_new_cars_fuel_plot <- renderPlotly({
    d <- data$new_cars_fuel %>%
      filter(Category == "New Private Cars") %>%
      filter(Fuel %in% names(fuel_colours)) %>%
      filter(Year < 2026)
    
    plot_ly(d,
            x = ~Year, y = ~Count,
            color = ~Fuel, colors = fuel_colours,
            type = "bar",
            hovertemplate = "<b>%{fullData.name}</b><br>%{x}: %{y:,.0f}<extra></extra>") %>%
      layout(
        barmode = "stack",
        xaxis = list(title = "Year", dtick = 1, showgrid = FALSE, showline = FALSE, ticks = "outside"),
        yaxis = list(title = "New private cars licensed",
                     gridcolor = "#e8ecf4", gridwidth = 1,
                     showline = FALSE, zeroline = FALSE),
        legend = list(orientation = "h", x = 0, y = -0.22),
        hovermode = "x unified"
      )
  })
  
  # 2. EV share of new private car registrations (derived)
  output$improve_ev_share_plot <- renderPlotly({
    d <- data$new_cars_fuel %>%
      filter(Category == "New Private Cars") %>%
      group_by(Year) %>%
      summarise(
        Total = sum(Count, na.rm = TRUE),
        Electric = sum(Count[Fuel == "Electric"], na.rm = TRUE),
        .groups = "drop"
      ) %>%
      mutate(EV_Pct = round(100 * Electric / Total, 1))
    
    plot_ly(d,
            x = ~Year, y = ~EV_Pct,
            type = "bar",
            marker = list(color = "#4dac26"),
            text = ~paste0(EV_Pct, "%"),
            textposition = "outside",
            hovertemplate = "<b>%{x}</b>: %{y:.1f}% electric<extra></extra>") %>%
      layout(
        xaxis = list(title = "Year", dtick = 1, showgrid = FALSE, showline = FALSE, ticks = "outside"),
        yaxis = list(title = "% of new private cars (BEV)",
                     gridcolor = "#e8ecf4", gridwidth = 1,
                     showline = FALSE, zeroline = FALSE),
        showlegend = FALSE
      )
  })
  
  # 3. New vs secondhand private cars by fuel type — grouped stacked bars
  output$improve_new_vs_secondhand_plot <- renderPlotly({
    d <- data$new_cars_fuel %>%
      filter(Fuel %in% names(fuel_colours)) %>%
      # Pick the latest year available so comparison is clearest
      filter(Year == max(Year))
    
    # Grouped horizontal stacked bar, two bars (New vs Secondhand)
    plot_ly(d,
            y = ~Category, x = ~Count,
            color = ~Fuel, colors = fuel_colours,
            type = "bar", orientation = "h",
            hovertemplate = "<b>%{fullData.name}</b><br>%{y}: %{x:,.0f}<extra></extra>") %>%
      layout(
        barmode = "stack",
        xaxis = list(title = paste0("Cars licensed (", max(d$Year), ")"),
                     gridcolor = "#e8ecf4", gridwidth = 1,
                     showline = FALSE, zeroline = FALSE),
        yaxis = list(title = "", showgrid = FALSE, showline = FALSE),
        legend = list(orientation = "h", x = 0, y = -0.25),
        margin = list(l = 150, r = 40)
      )
  })
  
  # 4. Vehicles licensed 1997-2024 long trend (TEA01)
  output$improve_long_trend_plot <- renderPlotly({
    d <- data$vehicles_licensed_long %>%
      filter(Type %in% c("New Private Cars", "New Goods Vehicles",
                         "Secondhand Private Cars", "New Motor Cycles"))
    
    colours <- c(
      "New Private Cars"         = "#405381",
      "Secondhand Private Cars"  = "#8693b3",
      "New Goods Vehicles"       = "#c0392b",
      "New Motor Cycles"         = "#00959f"
    )
    
    plot_ly(d,
            x = ~Year, y = ~Count,
            color = ~Type, colors = colours,
            type = "scatter", mode = "lines",
            line = list(width = 2),
            hovertemplate = "<b>%{fullData.name}</b><br>%{x}: %{y:,.0f}<extra></extra>") %>%
      layout(
        xaxis = list(title = "Year", showgrid = FALSE, showline = FALSE, ticks = "outside"),
        yaxis = list(title = "Vehicles licensed for first time",
                     gridcolor = "#e8ecf4", gridwidth = 1,
                     showline = FALSE, zeroline = FALSE),
        legend = list(orientation = "h", x = 0, y = -0.2),
        hovermode = "x unified"
      )
  })
  
  # 5. Private car fleet composition by fuel type (THA17)
  output$improve_fleet_fuel_plot <- renderPlotly({
    d <- data$car_fleet_fuel
    
    colours <- c(
      "Petrol"           = "#8693b3",
      "Diesel"           = "#405381",
      "Other fuel types" = "#4dac26"
    )
    
    plot_ly(d,
            x = ~Year, y = ~Count,
            color = ~Fuel, colors = colours,
            type = "bar",
            hovertemplate = "<b>%{fullData.name}</b><br>%{x}: %{y:,.0f} cars<extra></extra>") %>%
      layout(
        barmode = "stack",
        xaxis = list(title = "Year", dtick = 1, showgrid = FALSE, showline = FALSE, ticks = "outside"),
        yaxis = list(title = "Private cars in stock",
                     gridcolor = "#e8ecf4", gridwidth = 1,
                     showline = FALSE, zeroline = FALSE),
        legend = list(orientation = "h", x = 0, y = -0.22),
        hovermode = "x unified"
      )
  })
  
  # ========================================================================
  # OVERVIEW TAB — KPI INFOBOXES
  # ========================================================================

  # Headline: Private Car VKT vs 2019 baseline (CAP 2025 Avoid flagship)
  output$kpi_ev_share <- renderInfoBox({
    latest_vkt <- data$private_car_vkt %>% filter(Year == max(Year))
    base_vkt <- data$private_car_vkt %>% filter(Year == 2019) %>% pull(Total_Billion_Km)
    delta <- round((latest_vkt$Total_Billion_Km - base_vkt) / base_vkt * 100, 1)
    sign <- ifelse(delta >= 0, "+", "")
    infoBox(
      paste0("Private Car VKT (", latest_vkt$Year, ")"),
      paste0(round(latest_vkt$Total_Billion_Km, 1), " Bn km"),
      paste0(sign, delta, "% vs 2019 (CAP target: −20% by 2030)"),
      icon  = icon("road"),
      color = "orange"
    )
  })
  
  # Headline: EV share of new private cars (CAP 2025 Improve flagship)
  output$kpi_pt_journeys <- renderInfoBox({
    ev <- data$new_cars_fuel %>%
      filter(Category == "New Private Cars") %>%
      group_by(Year) %>%
      summarise(
        Total = sum(Count, na.rm = TRUE),
        Electric = sum(Count[Fuel == "Electric"], na.rm = TRUE),
        .groups = "drop"
      ) %>%
      mutate(EV_Pct = round(100 * Electric / Total, 1)) %>%
      filter(Year == max(Year))
    infoBox(
      paste0("New EV Share (", ev$Year, ")"),
      paste0(ev$EV_Pct, "%"),
      "Pure electric share of new private cars licensed",
      icon  = icon("bolt"),
      color = "green"
    )
  })
  
  # Headline: Remote working (Avoid behavioural lever)
  output$kpi_remote_work <- renderInfoBox({
    latest_rw <- data$remote_work %>% filter(Year == max(Year))
    infoBox(
      paste0("Remote Working (", latest_rw$Year, ")"),
      paste0(latest_rw$Remote_Work_Percent, "%"),
      "Workforce working at or from home",
      icon  = icon("house-laptop"),
      color = "blue"
    )
  })
